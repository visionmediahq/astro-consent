<script>
  document.querySelectorAll<HTMLElement>('[data-map]').forEach((wrap) => {
    const button = wrap.querySelector<HTMLButtonElement>('[data-map-load]')
    const facade = wrap.querySelector<HTMLElement>('[data-map-facade]')

    button?.addEventListener('click', () => {
      const iframe = document.createElement('iframe')
      iframe.src = wrap.dataset.src ?? ''
      iframe.title = wrap.dataset.title ?? 'Karta'
      iframe.loading = 'lazy'
      iframe.referrerPolicy = 'no-referrer-when-downgrade'
      iframe.setAttribute('allowfullscreen', '')
      facade?.remove()
      wrap.appendChild(iframe)
      // An iframe is not focusable on its own, so focus would have fallen back
      // to <body> and the keyboard user would have lost their place in the page.
      iframe.tabIndex = -1
      iframe.focus()
    })
  })
</script>
