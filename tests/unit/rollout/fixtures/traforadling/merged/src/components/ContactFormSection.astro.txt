<script>
  const RECAPTCHA_SITE_KEY = import.meta.env.PUBLIC_RECAPTCHA_SITE_KEY as string | undefined;

  // Load reCAPTCHA v3 script if configured
  if (RECAPTCHA_SITE_KEY) {
    const script = document.createElement('script');
    script.src = `https://www.google.com/recaptcha/api.js?render=${RECAPTCHA_SITE_KEY}`;
    script.async = true;
    document.head.appendChild(script);
  }

  const form = document.getElementById('contact-form-section') as HTMLFormElement | null;
  const successMsg = document.getElementById('form-success');
  const errorMsg = document.getElementById('form-error');
  const submitBtn = document.getElementById('form-submit-btn') as HTMLButtonElement | null;
  const btnLabel = document.getElementById('form-btn-label');
  const btnIcon = document.getElementById('form-btn-icon');

  async function getRecaptchaToken(): Promise<string> {
    if (!RECAPTCHA_SITE_KEY) return '';
    return new Promise<string>((resolve) => {
      (window as { grecaptcha?: { ready: (cb: () => void) => void; execute: (key: string, opts: { action: string }) => Promise<string> } }).grecaptcha?.ready(() => {
        (window as { grecaptcha?: { ready: (cb: () => void) => void; execute: (key: string, opts: { action: string }) => Promise<string> } }).grecaptcha!.execute(RECAPTCHA_SITE_KEY, { action: 'contact' }).then(resolve).catch(() => resolve(''));
      });
    });
  }

  if (form && submitBtn) {
    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      const data = new FormData(form);

      successMsg?.classList.add('hidden');
      errorMsg?.classList.add('hidden');

      submitBtn.disabled = true;
      if (btnLabel) btnLabel.textContent = 'Skickar…';
      if (btnIcon) (btnIcon as HTMLElement).style.display = 'none';

      try {
        const recaptchaToken = await getRecaptchaToken();

        const res = await fetch(form.action, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            name: data.get('name'),
            email: data.get('email'),
            message: data.get('message'),
            recaptchaToken,
          }),
        });

        if (res.ok) {
          successMsg?.classList.remove('hidden');
          form.reset();
        } else {
          errorMsg?.classList.remove('hidden');
        }
      } catch {
        errorMsg?.classList.remove('hidden');
      } finally {
        submitBtn.disabled = false;
        if (btnLabel) btnLabel.textContent = 'Skicka meddelande';
        if (btnIcon) (btnIcon as HTMLElement).style.display = '';
      }
    });
  }
</script>
