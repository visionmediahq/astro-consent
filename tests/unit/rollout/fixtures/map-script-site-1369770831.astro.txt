---
import { Phone, MapPin, ExternalLink, ArrowDown } from '@lucide/astro'
import Base from '../layouts/Base.astro'
import Header from '../components/Header.astro'
import Footer from '../components/Footer.astro'
import PageHero from '../components/PageHero.astro'
import ContactSection from '../components/ContactSection.astro'
import { company, schema } from '../data/site'

export const prerender = true

const address = `${company.address.streetAddress}, ${company.address.postalCode} ${company.address.addressLocality}`
const mapSrc = `https://maps.google.com/maps?q=${encodeURIComponent(address)}&z=15&output=embed`
---

<Base
  title={`Kontakta oss – ${company.name}`}
  description={`Kontakta Exempel Glas AB på ${address}. Ring ${company.phone.display} eller skicka ett meddelande så återkommer vi.`}
  canonical={`${company.url}/kontakt/`}
  ogImage={`${company.url}/og-image.jpg`}
  schema={schema}
>
  <Header />
  <main id="main">
    <PageHero
      title="Kontakta oss"
      lead="Kontakta oss idag för en kostnadsfri offert och låt oss hjälpa dig att förverkliga dina idéer med skräddarsydda och hållbara lösningar."
      image="glasarbete.jpg"
      imageAlt="Glasskiva som skärs till för hand"
      breadcrumbs={[{ label: 'Hem', href: '/' }, { label: 'Kontakt' }]}
    >
      <div class="mt-8 flex flex-wrap items-center gap-x-6 gap-y-3" data-animate data-animate-delay="0.15">
        <a href={company.phone.href} class="btn btn-primary btn-lg tabular-nums" onclick="window.umami?.track('phone_click')">
          <Phone size={18} stroke-width={2} aria-hidden="true" />
          {company.phone.display}
        </a>
        <a href="#kontaktformular" class="link-arrow" onclick="window.umami?.track('cta_click', { label: 'Till kontaktformuläret' })">
          Kontaktformulär
          <ArrowDown size={17} stroke-width={2} aria-hidden="true" />
        </a>
      </div>
    </PageHero>

    <ContactSection source="Kontaktsidan" showAddress>
      <div class="wrap mt-12 lg:mt-16" data-animate>
        <div
          class="relative h-[18rem] overflow-hidden rounded-box border border-base-300 bg-base-100 sm:h-[24rem]"
          data-map
          data-src={mapSrc}
          data-title={`Karta – ${address}`}
        >
          <div class="flex h-full flex-col items-center justify-center gap-4 px-6 text-center" data-map-placeholder>
            <span class="grid size-12 place-items-center rounded-full bg-base-200 text-accent">
              <MapPin size={24} stroke-width={2} aria-hidden="true" />
            </span>
            <p class="font-display text-h3">{address}</p>
            <p class="max-w-md text-base-content/75">
              Kartan visas via Google Maps, som kan sätta cookies. Den laddas först när du väljer att visa den.
            </p>
            <div class="flex flex-wrap items-center justify-center gap-x-6 gap-y-3">
              <button type="button" class="btn btn-primary btn-lg" data-map-load onclick="window.umami?.track('cta_click', { label: 'Visa karta' })">
                Visa karta
              </button>
              <a href={company.mapsUrl} target="_blank" rel="noopener noreferrer" class="link-arrow">
                Öppna i Google Maps
                <ExternalLink size={16} stroke-width={2} aria-hidden="true" />
                <span class="sr-only">(öppnas i ny flik)</span>
              </a>
            </div>
          </div>
        </div>
      </div>
    </ContactSection>
  </main>
  <Footer />
</Base>

<script>
  const map = document.querySelector<HTMLElement>('[data-map]')
  map?.querySelector('[data-map-load]')?.addEventListener('click', () => {
    const frame = document.createElement('iframe')
    frame.src = map.dataset.src ?? ''
    frame.title = map.dataset.title ?? 'Karta'
    frame.className = 'absolute inset-0 h-full w-full border-0'
    frame.referrerPolicy = 'no-referrer-when-downgrade'
    map.querySelector('[data-map-placeholder]')?.remove()
    map.append(frame)
    frame.focus()
  })
</script>
