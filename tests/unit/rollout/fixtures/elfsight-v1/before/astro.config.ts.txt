import { defineConfig } from 'astro/config'
import node from '@astrojs/node'
import sitemap from '@astrojs/sitemap'
import tailwindcss from '@tailwindcss/vite'

// Required for sitemap URL generation.
// import.meta.env reads from .env files (dev); process.env reads from system env (Docker/CI).
// Builder agent sets SITE_URL per client.
const site = import.meta.env.SITE_URL ?? process.env.SITE_URL ?? 'https://example.com'

// https://astro.build/config
export default defineConfig({
  site,

  output: 'server',

  security: {
    checkOrigin: false,
  },

  adapter: node({
    mode: 'standalone',
  }),

  integrations: [
    // In SSR mode pages are not auto-discovered. Pages here render date-dependent
    // content (opening hours, activity visibility) on every request, so they must
    // stay SSR rather than `prerender = true` — list them explicitly instead.
    sitemap({
      customPages: [
        `${site}/`,
        `${site}/meny/`,
        `${site}/produkter/`,
        `${site}/oppettider/`,
        `${site}/kontakta-oss/`,
      ],
    }),
  ],

  vite: {
    plugins: [tailwindcss()],
  },
})
