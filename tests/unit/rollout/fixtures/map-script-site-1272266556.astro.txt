<!-- Map (bottom of page)
     Click-to-load: embedding Google Maps directly sets Google cookies on every
     visitor before they have chosen anything. The map only loads once the
     visitor asks for it — which also keeps it off the critical render path. -->
<section aria-label={`Karta – ${company.address.streetAddress}, ${company.address.addressLocality}`} class="relative">
  <div

    data-map-src={`https://maps.google.com/maps?q=${mapsQuery}&z=14&output=embed`}
    data-map-title={`Karta – ${company.address.streetAddress}, ${company.address.postalCode} ${company.address.addressLocality}`}
  >
    <div class="map-placeholder">
      <span class="contact-row-icon"><MapPin size={22} aria-hidden="true" /></span>
      <p class="map-placeholder-title">{company.address.streetAddress}, {company.address.postalCode} {company.address.addressLocality}</p>
      <p class="map-placeholder-text">
        Kartan laddas från Google Maps. När du visar den kan Google spara cookies på din enhet.
      </p>
      <div class="mt-5 flex flex-col sm:flex-row gap-3 justify-center">
        <button type="button" id="map-load" class="btn-vm btn-vm--primary">Visa karta</button>
        <a
          href={`https://www.google.com/maps/search/?api=1&query=${mapsQuery}`}
          target="_blank"
          rel="noopener noreferrer"
          class="btn-vm btn-vm--ghost"
        >

        </a>
      </div>
    </div>
  </div>
</section>

<script>
  const mapEmbed = document.getElementById('map-embed');
  const mapButton = document.getElementById('map-load');

  mapButton?.addEventListener('click', () => {
    const src = mapEmbed?.dataset.mapSrc;
    if (!mapEmbed || !src) return;

    const iframe = document.createElement('iframe');
    iframe.src = src;
    iframe.title = mapEmbed.dataset.mapTitle ?? 'Karta';
    iframe.width = '100%';
    iframe.height = '420';
    iframe.loading = 'lazy';
    iframe.referrerPolicy = 'no-referrer-when-downgrade';
    iframe.style.border = '0';
    iframe.style.display = 'block';

    mapEmbed.replaceChildren(iframe);
    (window as unknown as { umami?: { track: (e: string, d?: unknown) => void } }).umami?.track('map_open');
  });
</script>
