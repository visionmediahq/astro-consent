---
import SEO from '../components/SEO.astro'
import SchemaOrg from '../components/SchemaOrg.astro'
import VisionFooter from '../components/VisionFooter.astro'
import '../styles/global.css'
import ConsentBanner from '@visionmediahq/astro-consent/components/ConsentBanner.astro'

interface Address {
  streetAddress: string
  addressLocality: string
  postalCode: string
  addressCountry: string
}

interface SchemaProps {
  name: string
  url: string
  telephone?: string
  email?: string
  address?: Address
  openingHours?: string[]
  description?: string
  image?: string
}

interface Props {
  // SEO — required on every page
  title: string
  description: string
  canonical: string
  ogImage?: string
  ogType?: string
  // LocalBusiness schema — omit on pages without business context (e.g. 404)
  schema?: SchemaProps
}

const { title, description, canonical, ogImage, ogType, schema } = Astro.props

const umamiScriptUrl = import.meta.env.UMAMI_SCRIPT_URL
const umamiWebsiteId = import.meta.env.UMAMI_WEBSITE_ID
const glitchtipDsn = import.meta.env.PUBLIC_GLITCHTIP_DSN
---

<!doctype html>
<html lang="sv">
  <head>
    <meta charset="utf-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1" />
    <link rel="icon" href="/favicon.webp" />

    <!-- Fonts: Barlow Condensed (headings) + Cabin (body) -->
    <link rel="preconnect" href="https://fonts.googleapis.com" />
    <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin />
    <link
      href="https://fonts.googleapis.com/css2?family=Barlow+Condensed:wght@600;700;800&family=Cabin:wght@400;600&display=swap"
      rel="stylesheet"
    />

    <SEO
      title={title}
      description={description}
      canonical={canonical}
      ogImage={ogImage}
      ogType={ogType}
    />

    {schema && (
      <SchemaOrg
        name={schema.name}
        url={schema.url}
        telephone={schema.telephone}
        email={schema.email}
        address={schema.address}
        openingHours={schema.openingHours}
        description={schema.description}
        image={schema.image}
      />
    )}

    <!-- GlitchTip error tracking — active when PUBLIC_GLITCHTIP_DSN is set -->
    {glitchtipDsn && (
      <>
        <script
          is:inline
          src="https://cdn.jsdelivr.net/npm/@sentry/browser@8/build/bundle.min.js"
          crossorigin="anonymous"
        ></script>
        <script is:inline define:vars={{ glitchtipDsn }}>
          if (typeof Sentry !== 'undefined') {
            Sentry.init({ dsn: glitchtipDsn, tracesSampleRate: 0.1 })
          }
        </script>
      </>
    )}

    <!-- Umami analytics — active when both UMAMI_SCRIPT_URL and UMAMI_WEBSITE_ID are set -->
    {umamiScriptUrl && umamiWebsiteId && (
      <script
        is:inline
        async
        src={umamiScriptUrl}
        data-website-id={umamiWebsiteId}
      ></script>
    )}

    <!-- Client theme tokens — overridden per client in index.astro -->
    <style is:global>
      :root {
        --color-primary:      var(--client-primary,    #e09900);
        --color-secondary:    var(--client-secondary,  #c08500);
        --color-accent:       var(--client-accent,     #c08500);
        --color-base-100:     var(--client-bg,         oklch(99% 0.008 80));
        --color-base-200:     var(--client-bg-2,       oklch(96% 0.01 80));
        --color-base-300:     var(--client-bg-3,       oklch(93% 0.01 80));
        --color-base-content: var(--client-text,       oklch(15% 0.015 60));
      }

      @media (prefers-reduced-motion: reduce) {
        *,
        *::before,
        *::after {
          animation-duration: 0.01ms !important;
          animation-iteration-count: 1 !important;
          transition-duration: 0.01ms !important;
          scroll-behavior: auto !important;
        }
      }

      /* client colors injected by builder.py */
      :root {
        --client-primary:   #e09900;
        --client-secondary: #c08500;
        --client-accent:    #c08500;
        --client-bg:        oklch(99% 0.008 80);
        --client-bg-2:      oklch(96% 0.01 80);
        --client-bg-3:      oklch(93% 0.01 80);
        --client-text:      oklch(15% 0.015 60);
      }
    </style>
  </head>
  <body class="bg-base-100 text-base-content">
    <a href="#main-content" class="skip-link">Hoppa till innehåll</a>
    <slot />
    <ConsentBanner />
    <VisionFooter />

    <!-- Global scroll-triggered entrance animations for [data-animate] elements -->
    <script>
      import { animate, inView } from 'motion'

      const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches

      if (!prefersReducedMotion) {
        inView('[data-animate]', (element) => {
          const raw = getComputedStyle(element as HTMLElement).getPropertyValue('--delay').trim()
          const delay = raw ? parseFloat(raw) : 0
          animate(
            element,
            { opacity: [0, 1], y: [20, 0] },
            { duration: 0.5, delay, easing: [0.22, 0.03, 0.26, 1] }
          )
        })
      }
    </script>
  </body>
</html>
