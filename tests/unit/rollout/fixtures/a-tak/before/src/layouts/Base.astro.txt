---
import SEO from '../components/SEO.astro'
import SchemaOrg from '../components/SchemaOrg.astro'
import VisionFooter from '../components/VisionFooter.astro'
import '../styles/global.css'

interface Address {
  streetAddress: string
  addressLocality: string
  postalCode: string
  addressCountry: string
}

interface SchemaProps {
  name: string
  url: string
  telephone?: string
  email?: string
  address?: Address
  openingHours?: string[]
  description?: string
  image?: string
}

interface Props {
  // SEO — required on every page
  title: string
  description: string
  canonical: string
  ogImage?: string
  ogType?: string
  // Preload hint for the above-the-fold hero image — improves LCP
  preloadImage?: string
  // LocalBusiness schema — omit on pages without business context (e.g. 404)
  schema?: SchemaProps
}

const { title, description, canonical, ogImage, ogType, preloadImage, schema } = Astro.props

const umamiScriptUrl = import.meta.env.UMAMI_SCRIPT_URL
const umamiWebsiteId = import.meta.env.UMAMI_WEBSITE_ID
const glitchtipDsn = import.meta.env.PUBLIC_GLITCHTIP_DSN
const recaptchaSiteKey = import.meta.env.PUBLIC_RECAPTCHA_SITE_KEY
---

<!doctype html>
<html lang="sv">
  <head>
    <meta charset="utf-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1" />
    <link rel="icon" href="/favicon.ico" sizes="32x32" />
    <link rel="icon" href="/favicon.webp" type="image/webp" sizes="192x192" />
    <link rel="apple-touch-icon" href="/apple-touch-icon.webp" />
    {preloadImage && <link rel="preload" as="image" href={preloadImage} fetchpriority="high" />}
    <link rel="preconnect" href="https://fonts.googleapis.com" />
    <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin />
    <link
      rel="stylesheet"
      href="https://fonts.googleapis.com/css2?family=Questrial&family=Red+Hat+Display:wght@400;500;600;700;800;900&display=swap"
      media="print"
      onload="this.media='all'"
    />
    <noscript>
      <link
        rel="stylesheet"
        href="https://fonts.googleapis.com/css2?family=Questrial&family=Red+Hat+Display:wght@400;500;600;700;800;900&display=swap"
      />
    </noscript>

    <SEO
      title={title}
      description={description}
      canonical={canonical}
      ogImage={ogImage}
      ogType={ogType}
    />

    {schema && (
      <SchemaOrg
        name={schema.name}
        url={schema.url}
        telephone={schema.telephone}
        email={schema.email}
        address={schema.address}
        openingHours={schema.openingHours}
        description={schema.description}
        image={schema.image}
      />
    )}

    <!-- GlitchTip error tracking — active when PUBLIC_GLITCHTIP_DSN is set -->
    {glitchtipDsn && (
      <>
        <script
          is:inline
          src="https://cdn.jsdelivr.net/npm/@sentry/browser@8/build/bundle.min.js"
          crossorigin="anonymous"
        ></script>
        <script is:inline define:vars={{ glitchtipDsn }}>
          if (typeof Sentry !== 'undefined') {
            Sentry.init({ dsn: glitchtipDsn, tracesSampleRate: 0.1 })
          }
        </script>
      </>
    )}

    <!-- reCAPTCHA V3 — active when PUBLIC_RECAPTCHA_SITE_KEY is set -->
    {recaptchaSiteKey && (
      <>
        <meta name="recaptcha-site-key" content={recaptchaSiteKey} />
        <script
          is:inline
          async
          src={`https://www.google.com/recaptcha/api.js?render=${recaptchaSiteKey}`}
        ></script>
      </>
    )}

    <!-- Umami analytics — active when both UMAMI_SCRIPT_URL and UMAMI_WEBSITE_ID are set -->
    {umamiScriptUrl && umamiWebsiteId && (
      <script
        is:inline
        async
        src={umamiScriptUrl}
        data-website-id={umamiWebsiteId}
      ></script>
    )}

    <!-- Client theme tokens — overridden per client in index.astro -->
    <style is:global>
      :root {
        --color-primary: var(--client-primary, #570df8);
        --color-secondary: var(--client-secondary, #f000b8);
        --color-accent: var(--client-accent, #37cdbe);
        --color-base-100: var(--client-bg, #ffffff);
        --color-base-content: var(--client-text, #1f2937);
      }

      @media (prefers-reduced-motion: reduce) {
        *,
        *::before,
        *::after {
          animation-duration: 0.01ms !important;
          animation-iteration-count: 1 !important;
          transition-duration: 0.01ms !important;
          scroll-behavior: auto !important;
        }
      }
    
      /* client colors injected by builder.py */
      :root {
        --client-primary:   #ff0000;
        --client-secondary: #1d2150;
        --client-bg:        #ffffff;
        --client-text:      #6c7176;
      }
    </style>
    <!-- Noscript: reveal all animated elements if JS is disabled -->
    <noscript>
      <style>[data-animate] { opacity: 1 !important; transform: none !important; }</style>
    </noscript>
  </head>
  <body class="bg-base-100 text-base-content">
    <slot />
    <VisionFooter />

    <!-- Global scroll-triggered entrance animations for [data-animate] elements -->
    <script>
      import { animate, inView } from 'motion'

      const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches

      if (!prefersReducedMotion) {
        inView('[data-animate]', (element) => {
          const el = element as HTMLElement
          if (el.dataset.animated) return
          el.dataset.animated = 'true'
          const delay = parseFloat(el.dataset.animateDelay ?? '0')
          animate(
            el,
            { opacity: [0, 1], y: [20, 0] },
            { duration: 0.5, delay, easing: [0.22, 0.03, 0.26, 1] }
          )
        })
      }
    </script>
  </body>
</html>
