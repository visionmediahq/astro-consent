<script define:vars={{ MAP_SRC }}>
  const btn = document.getElementById('map-accept-btn')
  const consent = document.getElementById('map-consent')
  const iframe = document.getElementById('map-iframe')

  function loadMap() {
    if (iframe) {
      iframe.src = MAP_SRC
      iframe.style.display = 'block'
    }
    consent?.remove()
  }

  btn?.addEventListener('click', () => {
    localStorage.setItem('maps-consent', '1')
    loadMap()
  })

  if (localStorage.getItem('maps-consent') === '1') {
    loadMap()
  }
</script>
