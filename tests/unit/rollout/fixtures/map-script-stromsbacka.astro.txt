      <button

        type="button"

        style="background: var(--color-primary);"
      >
        Visa karta
      </button>

    <!-- Iframe injected here on consent -->

<script>
  const btn = document.getElementById('map-consent-btn');
  const placeholder = document.getElementById('map-placeholder');
  const container = document.getElementById('map-container');

  btn?.addEventListener('click', () => {
    if (!container || !placeholder) return;

    const iframe = document.createElement('iframe');
    iframe.title = 'Karta – Strömsbacka Återvinning, Silvermyntsgatan 5, Sala';
    iframe.src = 'https://www.google.com/maps/embed?pb=!1m18!1m12!1m3!1d2043.7!2d16.6027!3d59.9211!2m3!1f0!2f0!3f0!3m2!1i1024!2i768!4f13.1!3m3!1m2!1s0x465e4f7a0cd1c9e5%3A0x1b4a13f3c5d4a9e7!2sSilvermyntsgatan%205%2C%20733%2037%20Sala!5e0!3m2!1ssv!2sse!4v1690000000000!5m2!1ssv!2sse';
    iframe.width = '100%';
    iframe.height = '420';
    iframe.style.cssText = 'border:0; display:block; width:100%; height:100%;';
    iframe.allowFullscreen = true;
    iframe.loading = 'lazy';
    iframe.referrerPolicy = 'no-referrer-when-downgrade';

    container.appendChild(iframe);
    container.classList.remove('hidden');
    placeholder.classList.add('hidden');
  });
</script>
